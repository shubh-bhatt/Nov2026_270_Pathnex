Day 04 — Basic CI/CD Concepts with Jenkins & GitLab


# my practice (Jenkinsfile - Basic CI Pipeline)

pipeline {
    agent any
    stages {
        stage('Build') {
            steps {
                echo 'Building project'
            }
        }
        stage('Test') {
            steps {
                echo 'Running tests'
            }
        }
        stage('Deploy') {
            steps {
                echo 'Deploying application'
            }
        }
    }
}




# my practice (GitLab CI/CD - Basic Pipeline)

stages:
	- build
	- test
	- deploy

build:
	stage: build
	script:
		- echo "Building project"

test:
	stage: test
	script:
		- echo "Running tests"

deploy:
	stage: deploy
	script:
		- echo "Deploying application"




# my practice (Ansible - Install and Configure Nginx)
- name: Install and configure Nginx
	hosts: all
	become: yes
	tasks:
		- name: Install Nginx
			yum:
				name: nginx
				state: present
		- name: Configure nginx
			template:
				src: nginx.conf.j2
				dest: /etc/nginx/nginx.conf
		- name: Start nginx service
			service:
				name: nginx
				state: started
				enables: yes



# my practice (Terraform - EC2 with Security Group)
resource "aws_security_group" "example" {
	name				= "example_security_group"
	description	=	"Allow SSH and HTTP"
	ingress {
		from_port		=	22
		to_port			= 22
		protocol		= "tcp"
		cidr_blocks	=	["0.0.0.0/0"]
	}

	ingress {
		from_port		= 80
		to_port			= 80
		protocol		= "tcp"
		cidr_blocks	= ["0.0.0.0/0"]
	}

	egress {
		from_port		= 0
		to_port			= 0
		protocol		= "-1"
		cidr_blocks	= ["0.0.0.0/0"]
	}
}
