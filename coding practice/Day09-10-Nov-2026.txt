Day 09 — Docker Integration with Jenkins & GitLab


# my practice (Ansible - setup docker container for nginx)
- name: Setup Docker Container for Nginx
	hosts: all
	become: yes
	tasks:
		- name: Pull Nginx image
			docker_image:
				name: nginx
				source: pull
		- name: Run Nginx container
			docker_container:
				name: nginx-container
				image: nginx
				state: started
				published_ports:
					-	"8080:80"

# (Kubernetes - Horizontal Pod Autoscaler (HPA))
apiVersion: apps/v1
kind: Deployment
metadata:
	name: pathnex-deployment
spec:
	replicas: 1
	selector:
		matchLabels:
			app: pathnex-app
	template:
		metadata:
			labels:
				app: pathnex-app
		spec:
			containers:
				- name: nginx
					image: nginx
					ports:
						- containerPort: 80

---

apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
	name: pathnex-hpa
spec:
	scaleTargetRef:
		apiVersion: apps/v1
		kind: Deployment
		name: pathnex-deployment
	minReplicas: 1
	maxReplicas: 10
	targetCPUUtilizationPercentage: 50


# Jenkisnfile - Deploy to Kubernetes
pipeline {
	agent any
	stages {
		stage('Deploy to Kubernetes') {
			steps {
				script {
					sh 'kubectl apply -f deployment.yaml'
				}
			}
		}
	}
}


# GitLab CI/CD - Deploy to Kubernetes
stages:
	- deploy

deploy:
	stage: deploy
	script:
		- kubectl apply -f kubernetes/deployment.yaml
