# my practice (Jenkinsfile - With Git integration and testing)

```groovy
pipeline {
    agent any
    stages {
        stage('Clone Repository') {
            steps {
                git 'https://github.com/shubh-bhatt/pokedexcli.git'
            }
        }
        stage('Build) {
            steps {
                echo 'Building project from Git repository'
            }
        }
        stage('Test') {
            steps {
                echo 'Running unit tests'
            }
        }
        stage('Deploy') {
            steps {
                echo 'Deploying application'
            }
        }
    }
}

# my practice (Ansible - Create User and set permissions)
- name: Create a new user
  hosts: all
  become: yes
  tasks:
    - name: Create user
      user:
        name: pathnex
        state: present
        shell: /bin/bash
    - name: Set permissions for the user
      file:
        path: /home/pathnex
        owner: pathnex
        group: pathnex
        mode: '0755'
